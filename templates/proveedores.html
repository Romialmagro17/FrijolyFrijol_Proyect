@app.route('/productos/nuevo', methods=['GET', 'POST']) @login_required def formulario_producto(): conexion = obtener_conexion() cursor = conexion.cursor() if request.method == 'POST': nombre = request.form.get('nombre') precio = float(request.form.get('precio')) if request.form.get('precio') else 0.0 stock = int(request.form.get('stock')) if request.form.get('stock') else 0 descripcion = request.form.get('descripcion', 'Sin descripción') id_proveedor = request.form.get('id_proveedor') # Capturamos el proveedor seleccionado cursor.execute( """ INSERT INTO productos (nombre, precio, stock, descripcion, id_proveedor) VALUES (%s, %s, %s, %s, %s) """, (nombre, precio, stock, descripcion, id_proveedor) ) conexion.commit() cursor.close() conexion.close() flash('Producto guardado exitosamente', 'success') return redirect(url_for('productos')) # Obtenemos la lista de proveedores para mostrarlos en un
